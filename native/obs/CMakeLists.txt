cmake_minimum_required(VERSION 3.22)
project(encore_capture LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

find_package(PkgConfig REQUIRED)
if(APPLE AND ENCORE_OBS_SOURCE AND ENCORE_OBS_DEPS AND ENCORE_OBS_RUNTIME)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/obsconfig.h" "#pragma once\n#define OBS_RELEASE_CANDIDATE 0\n#define OBS_BETA 0\n")
  add_library(obs_macos SHARED IMPORTED)
  set_target_properties(obs_macos PROPERTIES
    IMPORTED_LOCATION "${ENCORE_OBS_RUNTIME}/Contents/Frameworks/libobs.framework/libobs"
    INTERFACE_INCLUDE_DIRECTORIES "${ENCORE_OBS_SOURCE}/libobs;${ENCORE_OBS_DEPS}/include;${CMAKE_CURRENT_BINARY_DIR}")
  add_library(OBS::libobs ALIAS obs_macos)
  add_library(encore_ffmpeg INTERFACE)
  target_include_directories(encore_ffmpeg INTERFACE "${ENCORE_OBS_DEPS}/include")
  foreach(component avformat avcodec avutil)
    find_library(FF_${component} NAMES ${component} PATHS "${ENCORE_OBS_DEPS}/lib" NO_DEFAULT_PATH REQUIRED)
    target_link_libraries(encore_ffmpeg INTERFACE "${FF_${component}}")
  endforeach()
  set(CMAKE_BUILD_RPATH "${ENCORE_OBS_RUNTIME}/Contents/Frameworks;${ENCORE_OBS_DEPS}/lib")
  set(CMAKE_INSTALL_RPATH "@loader_path/../Contents/Frameworks")
  set(FFMPEG_TARGET encore_ffmpeg)
else()
  pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil)
  pkg_check_modules(LIBOBS REQUIRED IMPORTED_TARGET libobs)
  add_library(OBS::libobs ALIAS PkgConfig::LIBOBS)
  set(FFMPEG_TARGET PkgConfig::FFMPEG)
endif()

add_library(encore_capture SHARED engine.cpp mux.cpp diagnostic-sources.cpp)
target_link_libraries(encore_capture PRIVATE OBS::libobs ${FFMPEG_TARGET})
target_compile_options(encore_capture PRIVATE -Wall -Wextra -Wpedantic)
if(APPLE)
  enable_language(OBJCXX)
  target_sources(encore_capture PRIVATE platform-macos.mm)
  target_link_libraries(encore_capture PRIVATE "-framework AppKit" "-framework ApplicationServices" "-framework AVFoundation")
else()
  pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk+-3.0)
  pkg_check_modules(X11 REQUIRED IMPORTED_TARGET x11)
  pkg_check_modules(WAYLAND REQUIRED IMPORTED_TARGET wayland-client)
  pkg_check_modules(INDICATOR REQUIRED IMPORTED_TARGET ayatana-appindicator3-0.1)
  pkg_check_modules(PIPEWIRE REQUIRED IMPORTED_TARGET libpipewire-0.3)
  target_sources(encore_capture PRIVATE platform-linux.cpp wayland-shortcuts.cpp)
  target_link_libraries(encore_capture PRIVATE PkgConfig::GTK PkgConfig::X11 PkgConfig::WAYLAND PkgConfig::INDICATOR PkgConfig::PIPEWIRE)
endif()
install(TARGETS encore_capture LIBRARY DESTINATION lib)
